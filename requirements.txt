# sms.py
import africastalking
import os

# Initialize Africa's Talking using environment variables
username = os.environ.get('AFRICAS_TALKING_USERNAME', 'sandbox')
api_key = os.environ.get('AFRICAS_TALKING_API_KEY')

if not api_key:
    raise ValueError("No Africa's Talking API key found in environment variables")

africastalking.initialize(username, api_key)

sms = africastalking.SMS

def send_sms(to_number, message):
    try:
        response = sms.send(message=message, recipients=[to_number])
        print("SMS sent successfully:", response)
        return response
    except Exception as e:
        print("Error sending SMS:", e)
        return None
